/**
 * @dev Interface of the ERC20 standard as defined in the EIP.
 */
interface IERC20 {
    event Transfer(address indexed from, address indexed to, uint256 value);
    event Approval(address indexed owner, address indexed spender, uint256 value);
    function totalSupply() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address to, uint256 value) external returns (bool);
    function allowance(address owner, address spender) external view returns (uint256);
    function approve(address spender, uint256 value) external returns (bool);
    function transferFrom(address from, address to, uint256 value) external returns (bool);
}

/**
 * @dev Interface for the optional metadata functions from the ERC20 standard.
 */
interface IERC20Metadata is IERC20 {
    function name() external view returns (string memory);
    function symbol() external view returns (string memory);
    function decimals() external view returns (uint8);
}

/**
 * @dev Provides information about the current execution context, including the
 * sender of the transaction and its data.
 */
abstract contract Context {
    function _msgSender() internal view virtual returns (address) {
        return msg.sender;
    }
    function _msgData() internal view virtual returns (bytes calldata) {
        return msg.data;
    }
}

/**
 * @dev Contract module which provides a basic access control mechanism, where
 * there is an account (an owner) that can be granted exclusive access to
 * specific functions.
 */
abstract contract Ownable is Context {
    address private _owner;

    error OwnableUnauthorizedAccount(address account);
    error OwnableInvalidOwner(address owner);

    event OwnershipTransferred(address indexed previousOwner, address indexed newOwner);

    constructor(address initialOwner) {
        if (initialOwner == address(0)) {
            revert OwnableInvalidOwner(address(0));
        }
        _transferOwnership(initialOwner);
    }

    modifier onlyOwner() {
        _checkOwner();
        _;
    }

    function owner() public view virtual returns (address) {
        return _owner;
    }

    function _checkOwner() internal view virtual {
        if (owner() != _msgSender()) {
            revert OwnableUnauthorizedAccount(_msgSender());
        }
    }

    function renounceOwnership() public virtual onlyOwner {
        _transferOwnership(address(0));
    }

    function transferOwnership(address newOwner) public virtual onlyOwner {
        if (newOwner == address(0)) {
            revert OwnableInvalidOwner(address(0));
        }
        _transferOwnership(newOwner);
    }

    function _transferOwnership(address newOwner) internal virtual {
        address oldOwner = _owner;
        _owner = newOwner;
        emit OwnershipTransferred(oldOwner, newOwner);
    }
}

/**
 * @dev Contract module that helps prevent reentrant calls to a function.
 */
abstract contract ReentrancyGuard {
    uint256 private constant NOT_ENTERED = 1;
    uint256 private constant ENTERED = 2;
    uint256 private _status;

    error ReentrancyGuardReentrantCall();

    constructor() {
        _status = NOT_ENTERED;
    }

    modifier nonReentrant() {
        _nonReentrantBefore();
        _;
        _nonReentrantAfter();
    }

    function _nonReentrantBefore() private {
        if (_status == ENTERED) {
            revert ReentrancyGuardReentrantCall();
        }
        _status = ENTERED;
    }

    function _nonReentrantAfter() private {
        _status = NOT_ENTERED;
    }
}

/**
 * @title Gamblr
 * @dev Prediction Market for ARC Testnet.
 * 
 * ARC NETWORK NOTES:
 * - Native Gas Token: USDC (18 Decimals)
 * - ERC-20 Token Interface: USDC (6 Decimals)
 * 
 * To avoid precision errors, this contract strictly uses the ERC-20 interface.
 * Users pay Gas in Native USDC, but deposit/bet using the ERC-20 allowance system.
 */
contract Gamblr is Ownable, ReentrancyGuard {
    
    // --- Configuration ---
    IERC20 public immutable usdcToken;

    // We hardcode 6 decimals to align with the ARC USDC ERC-20 interface
    uint256 constant DECIMALS = 10**6; 

    // --- Fee & Limit Configuration ---
    // Min Deposit: 1 USDC
    uint256 public constant MIN_DEPOSIT = 1 * DECIMALS;
    // Min Stake: 0.1 USDC
    uint256 public constant MIN_STAKE = 1 * 10**5; 
    // Min Withdraw: 10 USDC
    uint256 public constant MIN_WITHDRAW = 10 * DECIMALS;
    
    // Fee: 1.0 USDC on Withdraw
    uint256 public constant FEE_WITHDRAW = 1 * DECIMALS; 
    // Fee: 0.1 USDC on Deposit and Bets
    uint256 public constant FEE_FLAT = 1 * 10**5;      

    // --- State Variables ---
    
    // Internal User Ledger (tracked in 6 decimals)
    mapping(address => uint256) public userBalances;
    
    // Accumulated fees (withdrawable by admin)
    uint256 public collectedFees;

    enum MarketOutcome { PENDING, YES, NO, VOID }

    struct Market {
        uint256 id;
        string question;
        uint256 totalYesAmount;
        uint256 totalNoAmount;
        MarketOutcome outcome;
        bool isResolved;
        uint256 endTime; // Unix timestamp
    }

    Market[] public markets;

    // Betting Ledger: marketId => user => amount
    mapping(uint256 => mapping(address => uint256)) public yesBets;
    mapping(uint256 => mapping(address => uint256)) public noBets;
    // Claim Tracking: marketId => user => bool
    mapping(uint256 => mapping(address => bool)) public hasClaimed;

    // --- Events ---
    event Deposit(address indexed user, uint256 amount, uint256 fee);
    event Withdraw(address indexed user, uint256 amount, uint256 fee);
    event MarketCreated(uint256 indexed marketId, string question, uint256 endTime);
    event BetPlaced(uint256 indexed marketId, address indexed user, bool isYes, uint256 amount, uint256 fee);
    event MarketResolved(uint256 indexed marketId, MarketOutcome outcome);
    event WinningsClaimed(uint256 indexed marketId, address indexed user, uint256 amount);

    /**
     * @param _usdcAddress The address of the USDC ERC-20 Interface on ARC.
     * IMPORTANT: Do not pass the native token precompile unless it adheres to 6 decimals.
     */
    constructor(address _usdcAddress) Ownable(msg.sender) {
        require(_usdcAddress != address(0), "Invalid token address");
        usdcToken = IERC20(_usdcAddress);
        
        // Optional Safety Check: Ensure the token explicitly returns 6 decimals
        try IERC20Metadata(_usdcAddress).decimals() returns (uint8 d) {
            require(d == 6, "Token must have 6 decimals for ARC USDC compatibility");
        } catch {
            // If decimals() is not implemented, we proceed with caution
        }
    }

    // --- Core User Actions ---

    /**
     * @dev Deposit USDC. 
     * Uses 'transferFrom'. User must Approve contract first.
     * Applies 0.1 USDC Fee.
     */
    function deposit(uint256 _amount) external nonReentrant {
        require(_amount >= MIN_DEPOSIT, "Deposit must be >= 1 USDC");
        
        // 1. Transfer USDC from user to this contract
        // We use the ERC-20 interface exclusively
        bool success = usdcToken.transferFrom(msg.sender, address(this), _amount);
        require(success, "USDC transfer failed. Check Allowance?");

        // 2. Calculate Fees
        uint256 fee = FEE_FLAT;
        
        // 3. Update Internal Ledger
        // If amount is exactly min_deposit (1.0), user gets 0.9
        uint256 amountAfterFee = _amount - fee;
        
        collectedFees += fee;
        userBalances[msg.sender] += amountAfterFee;

        emit Deposit(msg.sender, amountAfterFee, fee);
    }

    /**
     * @dev Place a bet.
     * Applies 0.1 USDC Fee from internal balance.
     */
    function placeBet(uint256 _marketId, bool _isYes, uint256 _stakeAmount) external nonReentrant {
        require(_marketId < markets.length, "Market not found");
        Market storage market = markets[_marketId];
        
        require(!market.isResolved, "Market resolved");
        require(block.timestamp < market.endTime, "Betting closed");
        require(_stakeAmount >= MIN_STAKE, "Stake too low (min 0.1 USDC)");

        // Total Deducted = Stake + 0.1 USDC Fee
        uint256 totalCost = _stakeAmount + FEE_FLAT;
        require(userBalances[msg.sender] >= totalCost, "Insufficient internal balance");

        // Deduct from balance
        userBalances[msg.sender] -= totalCost;
        collectedFees += FEE_FLAT;

        // Record Bet
        if (_isYes) {
            yesBets[_marketId][msg.sender] += _stakeAmount;
            market.totalYesAmount += _stakeAmount;
        } else {
            noBets[_marketId][msg.sender] += _stakeAmount;
            market.totalNoAmount += _stakeAmount;
        }

        emit BetPlaced(_marketId, msg.sender, _isYes, _stakeAmount, FEE_FLAT);
    }

    /**
     * @dev Claim winnings.
     * No fee is charged on claiming to ensure users can exit positions cleanly.
     */
    function claimWinnings(uint256 _marketId) external nonReentrant {
        Market storage market = markets[_marketId];
        require(market.isResolved, "Market not resolved");
        require(!hasClaimed[_marketId][msg.sender], "Already claimed");

        uint256 userBet;
        uint256 winningPool;
        uint256 losingPool;
        uint256 winnings = 0;

        if (market.outcome == MarketOutcome.YES) {
            userBet = yesBets[_marketId][msg.sender];
            winningPool = market.totalYesAmount;
            losingPool = market.totalNoAmount;
        } else if (market.outcome == MarketOutcome.NO) {
            userBet = noBets[_marketId][msg.sender];
            winningPool = market.totalNoAmount;
            losingPool = market.totalYesAmount;
        } else if (market.outcome == MarketOutcome.VOID) {
            userBet = yesBets[_marketId][msg.sender] + noBets[_marketId][msg.sender];
            winnings = userBet; // Refund
        }

        require(userBet > 0, "No winning bet");

        // Parimutuel Calculation
        if (market.outcome != MarketOutcome.VOID) {
            uint256 totalPool = winningPool + losingPool;
            // (UserBet * TotalPool) / WinningPool
            winnings = (userBet * totalPool) / winningPool;
        }

        hasClaimed[_marketId][msg.sender] = true;
        userBalances[msg.sender] += winnings;

        emit WinningsClaimed(_marketId, msg.sender, winnings);
    }

    /**
     * @dev Withdraw to external wallet.
     * Applies 1.0 USDC Fee.
     */
    function withdraw(uint256 _amount) external nonReentrant {
        require(_amount >= MIN_WITHDRAW, "Min withdraw is 10 USDC");
        require(userBalances[msg.sender] >= _amount, "Insufficient balance");

        // Deduct from internal ledger
        userBalances[msg.sender] -= _amount;

        // Calculate Fee
        uint256 fee = FEE_WITHDRAW;
        uint256 amountToSend = _amount - fee;
        collectedFees += fee;

        // Transfer via ERC-20 Interface
        bool success = usdcToken.transfer(msg.sender, amountToSend);
        require(success, "Transfer failed");

        emit Withdraw(msg.sender, amountToSend, fee);
    }

    // --- Admin ---

    function createMarket(string memory _question, uint256 _durationSeconds) external onlyOwner {
        markets.push();
        uint256 newId = markets.length - 1;
        Market storage m = markets[newId];
        m.id = newId;
        m.question = _question;
        m.outcome = MarketOutcome.PENDING;
        m.endTime = block.timestamp + _durationSeconds;
        
        emit MarketCreated(newId, _question, m.endTime);
    }

    function resolveMarket(uint256 _marketId, MarketOutcome _outcome) external onlyOwner {
        require(_marketId < markets.length, "Invalid ID");
        require(!markets[_marketId].isResolved, "Already resolved");
        require(_outcome != MarketOutcome.PENDING, "Outcome cannot be pending");

        markets[_marketId].outcome = _outcome;
        markets[_marketId].isResolved = true;

        emit MarketResolved(_marketId, _outcome);
    }

    function withdrawFees() external onlyOwner {
        uint256 amount = collectedFees;
        collectedFees = 0;
        require(usdcToken.transfer(msg.sender, amount), "Transfer failed");
    }
}
